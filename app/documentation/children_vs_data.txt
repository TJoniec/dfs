FUNCTION generate_card_layout(df, saved_uuids = ∅) -> Div:
  IF df is empty:
    RETURN Div("No lineups generated.")

  FOR row IN df.rows:
    uid := row["uuid-index"]

    header_text := f"Lineup #{row['Lineup ID']} — FPTS: {row['Total Lineup FPTS']}, Salary: {row['Total Lineup Salary']}"

    player_items := []
    FOR slot IN ['QB1','RB1','RB2','WR1','WR2','WR3','TE1','FLEX','DST1']:
      player := row.get(slot)
      IF player != 'None':
        player_items.APPEND( Li(f"{slot}: ...") )

    checkbox := Checkbox(
        id = {"type":"save-toggle","index": uid},
        value = (uid IN saved_uuids)  // reflect saved state
    )

    card := Card(
      Header( header_text, checkbox ),
      Body( Ul(player_items) ),
      id = {"type":"optimizer-lineup-card-initial-uuid","index": uid}
    )

    ADD card TO list_of_cards

  RETURN Div(list_of_cards)

----Save/Unsave toffle (works in both columns)
CALLBACK -> Output(saved-lineups-store.data)
INPUTS:
  values := Input( ALL save-toggle.value )   // list[bool], same order as ids state below
STATE:
  ids    := State( ALL save-toggle.id )      // list[dict] with {"index": uuid}
  saved  := State(saved-lineups-store.data)
  all    := State(global-store-df3.data)
BEHAVIOR:
  IF no ctx.triggered_id:
    PreventUpdate

  fired_uuid := ctx.triggered_id["index"]

  // find current value for the fired checkbox by matching its id
  idx := index i where ids[i]["index"] == fired_uuid
  current_value := values[idx]  // True = save, False = unsave

  IF current_value == True:
    lineup := FIND x IN all WHERE x["uuid-index"] == fired_uuid
    IF lineup exists AND fired_uuid NOT IN map(y -> y["uuid-index"], saved):
      saved.APPEND(lineup)
  ELSE:
    saved := [ y IN saved WHERE y["uuid-index"] != fired_uuid ]

  RETURN saved

  ----Invariants Initialization
  LAYOUT INCLUDES:
  dcc.Store(id="global-store-df3", data=[...])     // populated elsewhere
  dcc.Store(id="saved-lineups-store", data=[], storage_type="session" or "local")

REQUIREMENT:
  Every lineup receives a stable "uuid-index" once, at creation time.
